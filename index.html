Учёт часов и выплат

Две контрольные даты: 9 и 25 число месяца. Ставка привязана к должности, бонусы и штрафы учитываются отдельной строкой.

Настройки ставок

Введите почасовые тарифы по должностям. Любая смена с указанной должностью возьмёт соответствующую ставку.

Добавить смену

Фиксируйте дату, время, сотрудника и должность. Автоматически рассчитаем часы и оплату.

Список смен

Часы считаются автоматически (пересчёт при изменении ставок). Можно удалить строку, если внесли неверные данные.
# Дата Сотрудник Должность Начало Конец Часы Ставка Премия/штраф Оплата, ₽

Сводка выплат

Рассчитываем по контрольным датам: 9 (за период 26 числа предыдущего месяца по 9 включительно) и 25 (за период 10–25). Суммируем часы, оплату и бонусы/штрафы по каждому сотруднику.

Код для Google Apps Script

function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  const ss = SpreadsheetApp.openById(data.spreadsheetId);
  const sheet = ss.getSheetByName('Shifts') || ss.insertSheet('Shifts');
  const payoutsSheet = ss.getSheetByName('Payouts') || ss.insertSheet('Payouts');

  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['Дата', 'Сотрудник', 'Должность', 'Начало', 'Конец', 'Часы', 'Ставка', 'Бонус/штраф', 'Оплата']);
  }
  if (payoutsSheet.getLastRow() === 0) {
    payoutsSheet.appendRow(['Сотрудник', 'Дата выплаты', 'Часы', 'Бонус/штраф', 'Итого']);
  }

  const shiftRows = data.shifts.map(s => [s.date, s.name, s.role, s.start, s.end, s.hours, s.rate, s.bonus, s.pay]);
  const payoutRows = data.payouts.map(p => [p.name, p.payoutDate, p.hours, p.bonus, p.total]);

  if (shiftRows.length) sheet.getRange(sheet.getLastRow() + 1, 1, shiftRows.length, shiftRows[0].length).setValues(shiftRows);
  if (payoutRows.length) payoutsSheet.getRange(payoutsSheet.getLastRow() + 1, 1, payoutRows.length, payoutRows[0].length).setValues(payoutRows);

  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON)
    .setHeader('Access-Control-Allow-Origin', '*')
    .setHeader('Access-Control-Allow-Headers', 'Content-Type');
}
